cmake_minimum_required(VERSION 3.24)
project(MandelbulbStudio VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/glfw-3.4 EXCLUDE_FROM_ALL)
add_library(glad STATIC vendor/glad/src/gl.c)
target_include_directories(glad PUBLIC vendor/glad/include)
add_executable(mandelbulb src/main.cpp)
target_compile_definitions(mandelbulb PRIVATE GLFW_INCLUDE_NONE)
target_link_libraries(mandelbulb PRIVATE glad glfw)
if(MSVC)
  target_compile_options(mandelbulb PRIVATE /W4 /utf-8)
endif()
add_custom_target(copy_shaders ALL
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders" "$<TARGET_FILE_DIR:mandelbulb>/shaders")
add_dependencies(mandelbulb copy_shaders)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT mandelbulb)
set_property(TARGET mandelbulb PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:mandelbulb>")
